export function transcriptionErrorMessageKey({ transcriptionStatus, transcriptionErrorCode, canGenerate }):
  if (transcriptionStatus !== "ERROR") return "aiGenerationError";
  if (transcriptionErrorCode !== "QUOTA") return "aiTranscriptionFailed";
  return canGenerate ? "aiTranscriptionQuotaFailed" : "aiTranscriptionQuotaFailedPublic";
