apps/web/app/s/[videoId]/_components/transcription-error-message.ts:
  export function transcriptionErrorMessageKey({ transcriptionStatus,
    transcriptionErrorCode, canGenerate }) {
    if (transcriptionStatus !== "ERROR") return "aiGenerationError";
    if (transcriptionErrorCode !== "QUOTA") return "aiTranscriptionFailed";
    // Public share links reach strangers: only the owner is told the
    // account is out of money. Everyone else gets the same promise without
    // the billing fact.
    return canGenerate
      ? "aiTranscriptionQuotaFailed"
      : "aiTranscriptionQuotaFailedPublic";
  }

`canGenerate` is computed server-side in page.tsx (AuthorizedContent,
:780-814): `if (!canUseAI(user)) return false; if (!userId) return false;`
-- an anonymous guest has no `user`/`userId` at all, so `canGenerate` is
ALWAYS false for any anonymous viewer (and also false for a logged-in
viewer who is neither the video owner nor an org owner/admin) -> always
routed to the "...Public" calm variant. Rendered in
GenerateAiPanel.tsx:611-617, which also keeps the "retry" action button
owner-only ("The RETRY BUTTON stays owner-only -- it spends real Gemini
money", :609) separately from the message text.
